Starting Data Stream..."); syncDashboard(); } else { console.error("Access Denied: Not logged in as Admin."); alert("Please log in with your admin email."); } }); function syncDashboard() { // 1. Sync User Management db.collection("users").onSnapshot(snap => { const userBody = document.getElementById("userTableBody"); if (snap.empty) console.warn("Notice: 'users' collection is empty."); document.getElementById("stat-total-users").innerText = snap.size; userBody.innerHTML = ""; snap.forEach(doc => { const u = doc.data(); userBody.innerHTML += ` ${u.email || 'No Email'} ${u.country || 'Bahrain'} ${u.subscriptionStatus || 'Free'} `; }); }, err => console.error("Firestore User Error:", err)); // 2. Sync Self Publishing db.collection("contacts").onSnapshot(snap => { const contactBody = document.getElementById("contactTableBody"); contactBody.innerHTML = ""; snap.forEach(doc => { const d = doc.data(); contactBody.innerHTML += ` ${d.firstName || ''} ${d.lastName || ''}
${d.email} ${d.contentTitle || 'N/A'} ${d.description || 'No Description'} ${d.submittedAt ? d.submittedAt.toDate().toLocaleDateString() : 'New'} `; }); }, err => console.error("Firestore Contacts Error:", err)); // 3. Sync Sales Analytics db.collection("sales").onSnapshot(snap => { let total = 0; document.getElementById("stat-active").innerText = snap.size; snap.forEach(doc => { total += parseFloat(doc.data().amount || 0); }); document.getElementById("stat-revenue").innerText = `BHD ${total.toFixed(3)}`; }); }
